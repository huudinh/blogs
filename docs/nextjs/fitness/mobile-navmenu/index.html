Mobile Navigation Menu

Tiến hành thêm Mobile Navigation Menu vào website.

Ẩn/Hiện mục lục

1. Sửa ThemeToggle​

Sửa file src/components/ThemeToggle.tsx
"use client";
import { Button } from "@/components/ui/button";
import { Moon, Sun } from "lucide-react";
import { useTheme } from "next-themes";
import { useEffect, useState } from "react";

const ThemeToggle = () => {
    const { theme, setTheme } = useTheme();
    const [mounted, setMounted] = useState(false);

    useEffect(() => {
        setMounted(true);
    }, []);

    if (!mounted) return null;
    return (
        <Button
            variant="ghost"
            size="icon"
            onClick={() => setTheme(theme === "light" ? "dark" : "light")}
        >
            {theme === "light" ? <Sun className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
        </Button>
    );
};

export default ThemeToggle;

2. Tạo MobileNavigation.tsx​

Tạo file src/components/MobileNavigation.tsx
"use client";
import { useState } from "react";
import Link from "next/link";
import ThemeToggle from "./ThemeToggle";
import { Button } from "./ui/button";
import { Menu, X } from "lucide-react";
import { navItems } from "@/lib/constants";
import { useActiveSection } from '@/hooks/useActiveSection';

export default function MobileNavigation() {
    const [isOpen, setIsOpen] = useState(false);
    const active = useActiveSection();

    return (
        <>
            <div className="flex items-center gap-1">
                <ThemeToggle />
                <Button
                    variant="ghost"
                    size="icon"
                    className="size-8 rounded-md text-[#676767] cursor-pointer dark:text-[#9c9c9c]"
                    onClick={() => setIsOpen(!isOpen)}
                >
                    {isOpen ? (
                        <X className="size-7" />
                    ) : (
                        <Menu className="size-7" />
                    )}
                </Button>

            </div>

            {isOpen && (
                <div
                    className={`shadow-md fixed left-0 right-0 top-16 z-40 md:hidden
                        ${isOpen ? "opacity-100" : "opacity-0 pointer-events-none"}`}
                >
                    <div className="px-2 pt-2 pb-3 sp-y-1 sm:px-3 bg-background border-t border-border dark:border-secondary/40">
                        {navItems.map((item) => (
                            <Link
                                className={`
                                    block px-3 py-2 transition-colors
                                    ${active === item.id
                                        ? 'text-primary font-semibold'
                                        : 'text-foreground hover:text-primary font-medium'
                                    }
`}
                                onClick={() => setIsOpen(false)}
                                key={item.name} href={item.href}>
                                {item.name}
                            </Link>
                        ))}
                        <div className="px-3 py-2">
                            <Button className="w-full font-semibold">Join now</Button>
                        </div>
                    </div>
                </div>
            )}
        </>
    );
}

import ThemeToggle from "./ThemeToggle";

    const active = useActiveSection();

    return (
        <nav className="fixed top-0 left-0 right-0 z-50 bg-background/95 backdrop-blur-md border-b border-border transition-colors duration-300">
            <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">

                    {/* Destop navigation */}
                    <div className="hidden md:block">

                            {navItems.map((item) => (
                                <Link
                                    className={`
                                        block px-3 py-2 transition-colors
                                        ${active === item.id
                                            ? 'text-primary font-semibold'
                                            : 'text-foreground hover:text-primary font-medium'
                                        }`}
                                    key={item.name} href={item.href}>
                                    {item.name}
                                </Link>
                            ))}
                        </div>
                    </div>

                    {/* CTA */}
                    <div className="hidden md:flex items-center space-x-4">
                        <ThemeToggle />
                        <Button className="font-semibold">Join Now</Button>
                    </div>

                    {/* Mobile menu button */}
                    <div className="md:hidden">
                        <MobileNavigation />
                    </div>
                </div>
            </div>
        </nav>
    );
}

4. Tạo hook useActiveSection​

Tạo file src/hooks/useActiveSection.ts
"use client";

import { useEffect, useState } from 'react';

export const useActiveSection = (
    defaultSection = 'home'
) => {
    const [active, setActive] = useState(defaultSection);

    useEffect(() => {
        const sections = document.querySelectorAll<HTMLElement>('section[id]');

        if (!sections.length) return;

        const observer = new IntersectionObserver(
            (entries) => {
                const visibleSection = entries.find(
                    (entry) => entry.isIntersecting
                );

                if (visibleSection) {
                    setActive(visibleSection.target.id);
                }
            },
            {
                rootMargin: '-30% 0px -50% 0px',
                threshold: 0,
            }
        );

        sections.forEach((section) => {
            observer.observe(section);
        });

        return () => observer.disconnect();
    }, []);

    return active;
};
